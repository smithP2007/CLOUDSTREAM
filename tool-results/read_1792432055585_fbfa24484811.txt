   175→     * Gets the website captcha token
   176→     * discovered originally by https://github.com/ahmedgamal17
   177→     * optimized by https://github.com/justfoolingaround
   178→     *
   179→     * @param url the main url, likely the same website you found the key from.
   180→     * @param key used to fill https://www.google.com/recaptcha/api.js?render=....
   181→     *
   182→     * @param referer the referer for the google.com/recaptcha/api.js... request, optional.

   185→    // Try document.select("script[src*=https://www.google.com/recaptcha/api.js?render=]").attr("src").substringAfter("render=")
   186→    // To get the key
   187→    suspend fun getCaptchaToken(url: String, key: String, referer: String? = null): String? {
   188→        try {
   189→            val _url = Url(url)
   190→            val domain = base64Encode(
   191→                (_url.protocol.name + "://" + _url.host + ":443").encodeToByteArray(),
   192→            ).replace("\n", "").replace("=", ".")
   193→
   194→            val vToken =
   195→                app.get(
   196→                    "https://www.google.com/recaptcha/api.js?render=$key",
   197→                    referer = referer,
   198→                    cacheTime = 0
   199→                )
   200→                    .text
   201→                    .substringAfter("releases/")
   202→                    .substringBefore("/")
   203→            val recapToken =
   204→                app.get("https://www.google.com/recaptcha/api2/anchor?ar=1&hl=en&size=invisible&cb=cs3&k=$key&co=$domain&v=$vToken")
   205→                    .document
   206→                    .selectFirst("#recaptcha-token")?.attr("value")
   207→            if (recapToken != null) {
   208→                return app.post(
   209→                    "https://www.google.com/recaptcha/api2/reload?k=$key",
   210→                    data = mapOf(
   211→                        "v" to vToken,
   212→                        "k" to key,
   213→                        "c" to recapToken,
   214→                        "co" to domain,
   215→                        "sa" to "",
   216→                        "reason" to "q"
   217→                    ), cacheTime = 0
   218→                ).text
   219→                    .substringAfter("rresp\",\"")
   220→                    .substringBefore("\"")
   221→            }
   222→        } catch (e: Exception) {
   223→            logError(e)
   224→        }
   225→        return null
   226→    }

   237→    /**
   238→     * Get anime tracker information based on title, year and type.
   239→     * Both titles are attempted to be matched with both Romaji and English title.
   240→     * Uses the anilist api.
   241→     *
   242→     * @param titles uses first index to search, but if you have multiple titles and want extra guarantee to match you can also have that
   243→     * @param types Optional parameter to narrow down the scope to Movies, TV, etc. See TrackerType.getTypes()
   244→     * @param year Optional parameter to only get anime with a specific year
   245→     **/
   246→    suspend fun getTracker(
   247→        titles: List<String>,
   248→        types: Set<TrackerType>?,
   249→        year: Int?,
   250→        lessAccurate: Boolean
   251→    ): Tracker? {
   252→        return try {
   253→            require(titles.isNotEmpty()) { "titles must no be empty when calling getTracker" }
   254→
   255→            val mainTitle = titles[0]
   256→            val search =
   257→                trackerCache[mainTitle]
   258→                    ?: searchAnilist(mainTitle)?.also {
   259→                        trackerCache[mainTitle] = it
   260→                    } ?: return null
   261→
   262→            val res = search.data?.page?.media?.find { media ->
   263→                val matchingYears = year == null || media.seasonYear == year
   264→                val matchingTitles = media.title?.let { title ->
   265→                    titles.any { userTitle ->
   266→                        title.isMatchingTitles(userTitle)
   267→                    }
   268→                } ?: false
   269→
   270→                val matchingTypes = types?.any { it.name.equals(media.format, true) } == true
   271→                if (lessAccurate) matchingTitles || matchingTypes && matchingYears else matchingTitles && matchingTypes && matchingYears
   272→            } ?: return null
   273→
   274→            Tracker(
   275→                res.idMal,
   276→                null,
   277→                res.id.toString(),
   278→                res.coverImage?.extraLarge ?: res.coverImage?.large,
   279→                res.bannerImage
   280→            )
   281→        } catch (t: Throwable) {
   282→            logError(t)
   283→            null
   284→        }
   285→    }

   287→    /** Searches Anilist using title.
   288→     * @param title Title string of the show
   289→     * @return [AniSearch] data class holds search info.
   290→     * */
   291→    private suspend fun searchAnilist(
   292→        title: String?,
   293→    ): AniSearch? {
   294→        val query = """
   295→        query (
   296→          ${'$'}page: Int = 1
   297→          ${'$'}search: String
   298→          ${'$'}sort: [MediaSort] = [POPULARITY_DESC, SCORE_DESC]
   299→          ${'$'}type: MediaType
   300→        ) {
   301→          Page(page: ${'$'}page, perPage: 20) {
   302→            media(
   303→              search: ${'$'}search
   304→              sort: ${'$'}sort
   305→              type: ${'$'}type
   306→            ) {
   307→              id
   308→              idMal
   309→              title { romaji english }
   310→              coverImage { extraLarge large }
   311→              bannerImage
   312→              seasonYear
   313→              format
   314→            }
   315→          }
   316→        }
   317→    """.trimIndent().trim()
   318→
   319→        val data = mapOf(
   320→            "query" to query,
   321→            "variables" to mapOf(
   322→                "search" to title,
   323→                "sort" to "SEARCH_MATCH",
   324→                "type" to "ANIME",
   325→            )
   326→        ).toJson().toRequestBody(RequestBodyTypes.JSON.toMediaTypeOrNull())
   327→
   328→        return app.post("https://graphql.anilist.co", requestBody = data)
   329→            .parsedSafe<AniSearch>()
   330→    }
   331→}

   715→fun base64Decode(string: String): String {
   716→    // ISO-8859-1 decoding: each byte maps directly to its Unicode code point (0-255),
   717→    // so we mask each byte to unsigned and convert to the corresponding Char manually.
   718→    // decodeToString() can't be used here as it assumes UTF-8.
   719→    val bytes = base64DecodeArray(string)
   720→    return buildString(bytes.size) {
   721→        for (b in bytes) {
   722→            append((b.toInt() and 0xFF).toChar())
   723→        }
   724→    }
   725→}

   742→fun MainAPI.fixUrl(url: String): String {
   743→    if (url.startsWith("http") ||
   744→        // Do not fix JSON objects and arrays when passed as urls.
   745→        url.startsWith("{\"") || url.startsWith("[")
   746→    ) {
   747→        return url
   748→    }
   749→    if (url.isEmpty()) {
   750→        return ""
   751→    }
   752→
   753→    val startsWithNoHttp = url.startsWith("//")
   754→    if (startsWithNoHttp) {
   755→        return "https:$url"
   756→    } else {
   757→        if (url.startsWith('/')) {
   758→            return mainUrl + url
   759→        }
   760→        return "$mainUrl/$url"
   761→    }
   762→}

   773→/**
   774→ * Splits the parameters of a [Url] into a map of key-value pairs.
   775→ *
   776→ * Unlike a manual `split("&")` / `split("=")` implementation, this relies on Ktor's
   777→ * built-in parameters parser ([Url.parameters]), which already handles URL-decoding,
   778→ * malformed pairs, and parameters without a value.
   779→ *
   780→ * Note: if a parameter key appears multiple times (e.g. `?a=1&a=2`), only the **first**
   781→ * value is kept, since the return type is `Map<String, String>`. Use [Url.parameters]
   782→ * directly if you need all values for repeated keys.
   783→ *
   784→ * @param url the [Url] whose parameters should be extracted.
   785→ * @return a map of decoded parameter names to their first decoded value.
   786→ *
   787→ * @sample
   788→ * splitUrlParameters(Url("https://example.com/path?foo=bar&baz=qux"))
   789→ * // returns {"foo": "bar", "baz": "qux"}
   790→ */
   791→@Prerelease
   792→fun splitUrlParameters(url: Url): Map<String, String> {
   793→    return url.parameters.entries().associate { (key, values) -> key to values.firstOrNull().orEmpty() }
   794→}

   796→/**
   797→ * Splits the parameters of a raw URL [String] into a map of key-value pairs.
   798→ *
   799→ * Convenience overload for callers that have a URL as plain text rather than a parsed
   800→ * [Url] instance. Internally parses [url] with Ktor's [Url] constructor and delegates
   801→ * to [splitUrlParameters].
   802→ *
   803→ * @param url the URL string whose parameters should be extracted.
   804→ * @return a map of decoded parameter names to their first decoded value.
   805→ *
   806→ * @sample
   807→ * splitUrlParameters("https://example.com/path?foo=bar&baz=qux")
   808→ * // returns {"foo": "bar", "baz": "qux"}

   810→@Prerelease
   811→fun splitUrlParameters(url: String): Map<String, String> {
   812→    return splitUrlParameters(Url(url))

   911→/** This is the primary way to store score/rating. Use Score.from or Score.from10 to parse the score
   912→ * as it does not have a public constructor. Use toInt/toFloat to get back the score.
   913→ *
   914→ * Internally it stores it as an int up to 10^9 to represent up to 10 significant digits. So think
   915→ * of this as a decimal class specifically for ratings.
   916→ * */
   917→@JsonAutoDetect(fieldVisibility = JsonAutoDetect.Visibility.ANY)
   918→@Serializable
   919→class Score private constructor(
   920→    /** Decimal between [0, 10^9] representing the min score and max score respectively */
   921→    @JsonProperty("data") @SerialName("data") private val data: Int,
   922→) {
   923→    override fun hashCode(): Int = this.data.hashCode()
   924→    override fun equals(other: Any?): Boolean = other is Score && this.data == other.data
   925→
   926→    @Deprecated(
   927→        "toOld() is deprecated. Use other Score methods instead.",
   928→        level = DeprecationLevel.ERROR
   929→    )
   930→    fun toOld(): Int = toInt(10000)
   931→
   932→    fun toByte(maxScore: Int): Byte = toLong(maxScore).toByte()
   933→
   934→    fun toInt(maxScore: Int = 10): Int = toLong(maxScore).toInt()
   935→
   936→    fun toLong(maxScore: Int = 10): Long = (data.toLong() * maxScore.toLong()) / MAX.toLong()
   937→
   938→    fun toFloat(maxScore: Int = 10): Float =
   939→        (data.toFloat() / MAX.toFloat()) * maxScore.toFloat()
   940→
   941→    fun toDouble(maxScore: Int = 10): Double =
   942→        (data.toDouble() / MAX.toDouble()) * maxScore.toDouble()
   943→
   944→    override fun toString(): String = this.toString(10)
   945→
   946→    /** Formats the rating to a human readable format (with no rounding)
   947→     *
   948→     * However it may also return null if the score is less than the minimum score,
   949→     * this is to avoid 0.0/10.0 in case of default = 0
   950→     * */
   951→    @Throws(IllegalArgumentException::class)
   952→    fun toStringNull(
   953→        minScore: Double,
   954→        maxScore: Int,
   955→        decimals: Int = 1,
   956→        removeTrailingZeros: Boolean = true,
   957→        decimalChar: Char = '.'
   958→    ): String? {
   959→        if (toDouble() < minScore) return null
   960→        return toString(maxScore, decimals, removeTrailingZeros, decimalChar)
   961→    }
   962→
   963→    /** Formats the rating to a human readable format (with no rounding) */
   964→    @Throws(IllegalArgumentException::class)
   965→    fun toString(
   966→        maxScore: Int,
   967→        decimals: Int = 1,
   968→        removeTrailingZeros: Boolean = true,
   969→        decimalChar: Char = '.'
   970→    ): String {
   971→        require(maxScore in 1..1000) {
   972→            "maxScore ∈ [1,1000]"
   973→        }
   974→        require(decimals in 0..MAX_ZEROS) {
   975→            "decimals ∈ [0,$MAX_ZEROS]"
   976→        }
   977→        var number = data.toLong() * maxScore.toLong()
   978→        val chars = CharArray(MAX_ZEROS + 6)
   979→
   980→        for (i in chars.indices) {
   981→            chars[i] = (number % 10L).toInt().digitToChar()
   982→            number /= 10L
   983→        }
   984→
   985→        var trailingZeros = MAX_ZEROS - decimals
   986→        for (i in (MAX_ZEROS - decimals) until chars.size) {
   987→            if (chars[i] != '0') {
   988→                break
   989→            }
   990→            trailingZeros += 1
   991→        }
   992→
   993→        var leadingZeros = 0
   994→        for (i in chars.indices.reversed()) {
   995→            if (chars[i] != '0') {
   996→                break
   997→            }
   998→            leadingZeros += 1
   999→        }
  1000→
  1001→        val stringBuilder = StringBuilder()
  1002→        for (i in maxOf(MAX_ZEROS, (chars.size - leadingZeros - 1)) downTo MAX_ZEROS) {
  1003→            stringBuilder.append(chars[i])
  1004→        }
  1005→
  1006→        val end = if (removeTrailingZeros) {
  1007→            maxOf(MAX_ZEROS - decimals, trailingZeros)
  1008→        } else {
  1009→            MAX_ZEROS - decimals
  1010→        }
  1011→
  1012→        if (end <= MAX_ZEROS - 1) {
  1013→            stringBuilder.append(decimalChar)
  1014→            for (i in MAX_ZEROS - 1 downTo end) {
  1015→                stringBuilder.append(chars[i])
  1016→            }
  1017→        }
  1018→
  1019→        return stringBuilder.toString()
  1020→    }

  1371→/**
  1372→ * For APIs using the mainUrl in the prefix for `MainAPI::load`,
  1373→ * this function replaces the `scheme`, `host` and `port` from an old link to the new mainUrl.
  1374→ *
  1375→ * https://en.wikipedia.org/wiki/Uniform_Resource_Identifier
  1376→ * ```text
  1377→ *           userinfo       host      port
  1378→ *           ┌──┴───┐ ┌──────┴──────┐ ┌┴─┐
  1379→ *   https://john.doe@www.example.com:1234/forum/questions/?tag=networking&order=newest#:~:text=whatever
  1380→ *   └─┬─┘   └─────────────┬─────────────┘└───────┬───────┘ └────────────┬────────────┘ └───────┬───────┘
  1381→ *   scheme            authority                path                   query                 fragment
  1382→ * ```
  1383→ */
  1384→fun MainAPI.updateUrl(url: String): String {
  1385→    return try {
  1386→        val original = Url(url)
  1387→        val updated = Url(mainUrl)
  1388→
  1389→        URLBuilder().apply {
  1390→            takeFrom(updated)
  1391→            user = original.user
  1392→            password = original.password
  1393→            encodedPath = original.encodedPath
  1394→            fragment = original.fragment
  1395→
  1396→            parameters.clear()
  1397→            parameters.appendAll(original.parameters)
  1398→        }.buildString()
  1399→    } catch (t: Throwable) {
  1400→        logError(t)
  1401→        url
  1402→    }
  1403→}
